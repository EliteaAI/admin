# 3a: Users with prompt_lib_public -> ensure they have viewer
INSERT INTO public.auth_core__project_user_role (project_id, user_id, role_id)
SELECT pur.project_id, pur.user_id, viewer_role.id
FROM public.auth_core__project_user_role pur
JOIN public.auth_core__project_role custom_role
    ON pur.role_id = custom_role.id AND pur.project_id = custom_role.project_id
JOIN public.auth_core__project_role viewer_role
    ON viewer_role.project_id = pur.project_id AND viewer_role.name = 'viewer'
WHERE custom_role.name = 'prompt_lib_public'
AND NOT EXISTS (
    SELECT 1 FROM public.auth_core__project_user_role existing
    WHERE existing.project_id = pur.project_id
    AND existing.user_id = pur.user_id
    AND existing.role_id = viewer_role.id
);

# 3b: Users with public_admin -> ensure they have admin
INSERT INTO public.auth_core__project_user_role (project_id, user_id, role_id)

# Issue #5262: provision one platform-owned, non-expiring access token named
# 'elitea-system' for every user that lacks one. Without it the Support Assistant,
# internal MCP toolkits, code nodes and scheduled agent runs fail for users who never
# created a PAT (pat_missing / "User token not found. Please create user_token").
# Per-user, not per-project, so there is no project_id param.
# Idempotent: each insert is guarded by the partial unique index below, so a re-run,
# or a login provisioning the same user mid-run, inserts nothing extra.
# New users get the token at creation time; this task only backfills accounts that
# already existed. The task now fails loudly - a run that reports success really did
# reach every user. Suspended users are skipped: they cannot authenticate, and they
# get the token on their first login after being reinstated.
# The schema change is not here: auth_core alembic revision 202609161200 runs at
# pylon_auth boot. It renames any pre-existing token that already carried the reserved
# name to 'elitea-system-conflict-<id>' (uuid and expiry untouched, so it stays an
# ordinary token of its owner) and then adds a partial unique index enforcing one
# system token per user.
# Pre-flight, after pylon_auth is up - expect 0 rows. Any row listed is a token some
# user created under a name that is now reserved, renamed out of the way with its uuid
# and expiry intact: it still authenticates, is listed to its owner and is deletable by
# them. Do not delete these rows. A null expires does not tell you the row was the
# platform's - users could create non-expiring tokens under any name before the
# reservation existed - so there is no way to tell provenance from this query, and
# deleting on that guess revokes a working credential. Report the renames to the
# owners and leave the rows alone.
SELECT id, user_id, name, expires FROM public.auth_core__token
  WHERE name LIKE 'elitea-system-conflict-%';
